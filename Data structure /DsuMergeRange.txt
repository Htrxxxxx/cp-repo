 
struct DSU {
    vector<int> parent, sz, nxt;
    int cmp;

    DSU(int n) {
        parent.resize(n + 2);
        sz.assign(n + 2, 1);
        nxt.resize(n + 2);

        iota(parent.begin(), parent.end(), 0);
        iota(nxt.begin(), nxt.end(), 0);

        cmp = n;
    }

    int find(int u) {
        return parent[u] == u ? u : parent[u] = find(parent[u]);
    }

    bool merge(int u, int v) {
        u = find(u);
        v = find(v);

        if (u == v)
            return false;

        if (sz[u] < sz[v])
            swap(u, v);

        parent[v] = u;
        sz[u] += sz[v];
        --cmp;

        return true;
    }

    int getNext(int x) {
        return nxt[x] == x ? x : nxt[x] = getNext(nxt[x]);
    }

    void mergeRange(int l, int r) {
        for (int x = getNext(l); x < r; x = getNext(x)) {
            merge(x, x + 1);
            nxt[x] = getNext(x + 1);
        }
    }
};
