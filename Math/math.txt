// return i ^ -1
long long inv[MAX_N];
void precompute_inverses() {
    inv[1] = 1;
    for (int i = 2; i < MAX_N; i++) {
        inv[i] = (mod - mod / i) * inv[mod % i] % mod;
    }
}

// Returns minimum x for which a ^ x % m = b % m, a and m are coprime.
int psgs(int a, int b, int m) {
    a %= m, b %= m;
    int n = sqrt(m) + 1;

    int an = 1;
    for (int i = 0; i < n; ++i)
        an = (an * 1ll * a) % m;

    unordered_map<int, int> vals;
    for (int q = 0, cur = b; q <= n; ++q) {
        vals[cur] = q;
        cur = (cur * 1ll * a) % m;
    }

    for (int p = 1, cur = 1; p <= n; ++p) {
        cur = (cur * 1ll * an) % m;
        if (vals.count(cur)) {
            int ans = n * p - vals[cur];
            return ans;
        }
    }
    return -1;
}

// To calculate spf for first N integers .
vector<int> calculateSPF() {
    vector<int> spf(N);
    iota(spf.begin(), spf.end(), 0);

    for (int i = 2; i < N; ++i) {
        if (spf[i] == i) { // i is a prime number
            for (int j = i; j < N; j += i) {
                if (spf[j] == j) {
                    spf[j] = i;
                }
            }
        }
    }

    return spf;
}

//Using spf to get primes in log(n) + preprocessing (O(nlog(log(n)) .

vector<int> calculateSPF() {
    vector<int> spf(N);
    iota(spf.begin(), spf.end(), 0);

    for (int i = 2; i < N; ++i) {
        if (spf[i] == i) { // i is a prime number
            for (int j = i; j < N; j += i) {
                if (spf[j] == j) {
                    spf[j] = i;
                }
            }
        }
    }

    return spf;
}
vector<int> find_divisors_inlog (int n , vector<int> &spf) {
    vector<int> divisors ;
    while(spf[n] != n) {
        int d1 = spf[n] , d2 = n / spf[n] ;
        divisors.push_back(d1) ;
        if(d1 != d2)  // unique divisors
        divisors.push_back(d2) ;
        n /= spf[n] ;
    }
    return divisors ;
}

// time complexity O(sqrt(n))
int phi(int n) {
    int result = n;
    for (int i = 2; i * i <= n; i++) {
        if (n % i == 0) {
            while (n % i == 0)
                n /= i;
            result -= result / i;
        }
    }
    if (n > 1)
        result -= result / n;
    return result;
}
// O(n loglog(n)) .
void phi_1_to_n(int n) {
    vector<int> phi(n + 1);
    for (int i = 0; i <= n; i++)
        phi[i] = i;

    for (int i = 2; i <= n; i++) {
        if (phi[i] == i) {
            for (int j = i; j <= n; j += i)
                phi[j] -= phi[j] / i;
        }
    }
}
